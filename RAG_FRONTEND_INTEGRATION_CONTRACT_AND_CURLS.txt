4. TYPESCRIPT INTERFACES (DUNG CHO D-TS APP REACT NATIVE & WEB)
--------------------------------------------------------------------------------

export interface RAGChatRequest {
  message: string;
  contextType?: 'THEORY_QUESTION' | 'THEORY_SKILL' | 'SIMULATION_SCENARIO' | 'GENERAL';
  contextRefId?: string;
  licenseClass?: string;
}

export interface LegalSourceItem {
  sourceId: string;
  documentTitle: string;
  documentNumber: string;
  chapter: string;
  article: string;
  clause: string;
  point?: string;
  citation: string;
  snippet: string;
  validityStatus: 'VALID' | 'EXPIRED';
}

export interface QuestionOption {
  label: string;
  text: string;
}

export interface QuestionDetailItem {
  questionId: number;
  sourceQuestionId: number;
  chapter: number;
  questionText: string;
  options: QuestionOption[];
  correctAnswer: string;
  isCritical: boolean;
  explanation: string;
  skillCode?: string;
  skillName?: string;
}

export interface RAGChatResponse {
  answer: string;
  sources: LegalSourceItem[];
  contextUsed: string;
  remainingDailyQueries: number;
  isFallback: boolean;
  suggestedActions: string[];
  questionDetail?: QuestionDetailItem;
}

export interface RAGQuotaResponse {
  userId: string;
  remainingDailyQueries: number;
  maxDailyQueries: number;
  rateLimitPerMinute: number;
}

5. MA NGUON HOOK MAU TICH HOP (React / React Native useRAGAssistant.ts)
--------------------------------------------------------------------------------

import { useState } from 'react';
import axios from 'axios';

const RAG_BASE_URL = 'http://103.20.96.85:8091/api/learning-intelligence/v1/rag';

export const useRAGAssistant = (currentUserId: string) => {
  const [loading, setLoading] = useState<boolean>(false);
  const [response, setResponse] = useState<RAGChatResponse | null>(null);
  const [error, setError] = useState<string | null>(null);

  const askAssistant = async (
    question: string,
    contextType: 'THEORY_QUESTION' | 'THEORY_SKILL' | 'GENERAL' = 'GENERAL',
    contextRefId?: string,
    licenseClass: string = 'B'
  ) => {
    setLoading(true);
    setError(null);
    try {
      const res = await axios.post<RAGChatResponse>(
        `${RAG_BASE_URL}/chat`,
        {
          message: question,
          contextType,
          contextRefId,
          licenseClass,
        },
        {
          headers: {
            'Content-Type': 'application/json',
            'X-User-Id': currentUserId,
          },
        }
      );
      setResponse(res.data);
      return res.data;
    } catch (err: any) {
      if (err.response?.status === 429) {
        setError('Ban da gui qua nhanh (toi da 5 cau/phut) hoac da het han muc 30 cau/ngay!');
      } else {
        setError(err.response?.data?.message || 'Loi ket noi den Tro ly AI.');
      }
      throw err;
    } finally {
      setLoading(false);
    }
  };

  return { askAssistant, response, loading, error };
};

--------------------------------------------------------------------------------
6. MA LOI HTTP STATUS CAN BAT (ERROR HANDLING)
--------------------------------------------------------------------------------
- 200 OK: Thanh cong, hien thi cau tra loi + nguon luat + chi tiet cau hoi.
- 400 Bad Request: Thieu header X-User-Id hoac cau hoi vuot qua 500 ky tu.
- 429 Too Many Requests: Vuot qua 5 cau/phut hoac vuot qua 30 cau/ngay.
  -> Frontend show popup/toast: "Ban da het 30 luot hoi mien phi hom nay. Vui long quay lai vao ngay mai!"
- 500 Server Error: Loi may chu noi bo.
